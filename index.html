def precedence (op) :
    if op == '^':
        return 3
    elif op == '*' or op == '/':
        return 2
    elif op == '+' or op == '-':
        return 1
    else:
        return 0
#Function to convert infix to postfix
def infix_to_postfix (expression) :
    stack = []
    postfix =  ""
    for ch in expression:
        # Operand
        if ch.isalnum():
            postfix += ch
        # Opening bracket
        elif ch == '(':
            stack.append(ch)
        # Closing bracket
        elif ch == ')':
            while stack and stack[-1] != '(':
                postfix += stack.pop()
            stack.pop()
        # Operator
        else:
            while (stack and stack[-1] != '(' and
                    precedence (stack[-1]) >= precedence (ch)):
                postfix += stack.pop()
            stack. append(ch)
    # Pop remaining operators
    while stack:
        postfix += stack.pop()
    return postfix

postfix = infix_to_postfix(infix)

Experiment 8 — Linked List

# Node class
class Node:
    def __init__(self, data):
        self.data = data
        self.next = None
# Linked List class
class LinkedList:
    def __init__(self):
        self.head = None
    # Insert at beginning
    def insert_beginning(self, data):
        new_node = Node(data)
        new_node.next = self.head
        self.head = new_node
        print("Node inserted at beginning")
    # Insert at end
    def insert_end(self, data):
        new_node = Node(data)
        if self.head is None:
            self.head = new_node
            print("Node inserted at end")
            return
        temp = self.head
        while temp.next is not None:
            temp = temp.next
        temp.next = new_node
        print("Node inserted at end")
    # Delete from beginning
    def delete_beginning(self):
        if self.head is None:
            print("Linked list is empty")
            return
        self.head = self.head.next
        print("Node deleted from beginning")
    # Delete from end
    def delete_end(self):
        if self.head is None:
            print("Linked list is empty")
            return
        if self.head.next is None:
            self.head = None
            print("Node deleted from end")
            return
        temp = self.head
        while temp.next.next is not None:
            temp = temp.next
        temp.next = None
        print("Node deleted from end")
    # Display
    def display(self):
        if self.head is None:
            print("Linked list is empty")
            return
        temp = self.head
        while temp is not None:
            print(temp.data, end=" -> ")
            temp = temp.next
        print("None")
# Main program
linked_list = LinkedList()
while True:

    print("1. Insert at beginning")
    print("2. Insert at end")

    print("4. Delete from end")
    print("5. Display")
    print("6. Exit")
    choice = int(input("Enter your choice: "))
    if choice == 1:
        data = int(input("Enter data: "))
        linked_list.insert_beginning(data)
    elif choice == 2:
        data = int(input("Enter data: "))
        linked_list.insert_end(data)
    elif choice == 3:

Experiment 9 — Binary Search Tree (BST)

class Node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None
def insert(root, value):
    if root is None:
        return Node(value)
    if value < root.data:
        root.left = insert(root.left, value)
    elif value > root.data:
        root.right = insert(root.right, value)
    return root
def display(root, space=0):
    if root is None:
        return
    space += 5
    display(root.right, space)
    print()
    print(" " * space + str(root.data))
    display(root.left, space)
def inorder(root):
    if root is not None:
        inorder(root.left)
        print(root.data, end=" ")
        inorder(root.right)
def preorder(root):
    if root is not None:
        print(root.data, end=" ")
        preorder(root.left)
        preorder(root.right)
def postorder(root):
    if root is not None:
        postorder(root.left)
        postorder(root.right)
        print(root.data, end=" ")
# Main program
root = None
n = int(input("How many nodes you want to insert? "))
for i in range(n):
    value = int(input("Enter the value to insert: "))
    root = insert(root, value)
print("\nConstructed Binary Search Tree\n")
display(root)
print("\n\nInorder Traversal")
inorder(root)
print("\n\nPreorder Traversal")
preorder(root)
print("\n\nPostorder Traversal")
postorder(root)
print()

Experiment 12 — N Queens Problem

def is_safe(board, row, col, n):
    # Check column
    for i in range(row):
        if board[i] == col:
            return False
    # Check diagonals
    for i in range(row):
        if abs(board[i] - col) == abs(i - row):
            return False
    return True
def solve(board, row, n):
    # All queens are placed
    if row == n:
        print_board(board, n)
        return
    # Try every column
    for col in range(n):
        if is_safe(board, row, col, n):
            board[row] = col       # Place queen
            solve(board, row + 1, n) # Move to next row
            board[row] = -1        # Backtrack
def print_board(board, n):
    for i in range(n):
        for j in range(n):
            if board[i] == j:
                print("Q", end=" ")
            else:
                print(".", end=" ")
        print()
    print()
# Main program
n = int(input("Enter value of N: "))
board = [-1] * n
solve(board, 0, n)
